Función de búsqueda inteligente y flexible function buscarDestino() { const query = document.getElementById('searchInput').value.toLowerCase().trim(); if (!query) return; let encontrado = false; kmlLayer.eachLayer(function(layer) { if (layer.feature && layer.feature.properties && layer.feature.properties.name) { const nombreLugar = layer.feature.properties.name.toLowerCase(); // Compara si el texto escrito está contenido en el nombre (ej. buscar "cc1" encuentra "aula-cc1") if (nombreLugar.includes(query) || query.includes(nombreLugar)) { encontrado = true; // Centrar el mapa en el lugar encontrado if (layer.getBounds) { map.fitBounds(layer.getBounds(), { maxZoom: 19 }); } else if (layer.getLatLng) { map.setView(layer.getLatLng(), 19); } // Mostrar la descripción original del KML (con las imágenes de WhatsApp) let contenidoPopup = `${layer.feature.properties.name}`; if (layer.feature.properties.description) { contenidoPopup += `
${layer.feature.properties.description}
`; } layer.bindPopup(contenidoPopup).openPopup(); // Actualizar el panel inferior de información document.getElementById('infoPanel').style.display = 'block'; document.getElementById('rutaTitulo').innerText = `Destino: ${layer.feature.properties.name}`; document.getElementById('rutaDetalle').innerText = `Ubicación encontrada y seleccionada en el campus.`; } } }); if (!encontrado) { alert('No se encontró ningún lugar con ese nombre. Prueba escribiendo: "CC1", "Cajas", "Biblioteca", "A1", etc.'); } }
